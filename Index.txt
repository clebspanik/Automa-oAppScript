<script>
  let selectedPeriods = [];

  function fetchAvailability() {
    const dataVal = document.getElementById('data').value;
    const resDiv = document.getElementById('results');
    if (!dataVal) return;

    resDiv.innerHTML = '<div class="loading">🔍 Consultando agenda da escola...</div>';
    selectedPeriods = [];
    validate();

    google.script.run
      .withSuccessHandler(renderPeriods)
      .withFailureHandler(err => {
        Swal.fire('Erro', 'Não foi possível carregar a agenda.', 'error');
      })
      .verificarDisponibilidade(dataVal);
  }

  function renderPeriods(periods) {
    const resDiv = document.getElementById('results');
    resDiv.innerHTML = '';

    if (!periods || periods.length === 0) {
      resDiv.innerHTML = '<p style="text-align:center; color:red;">⚠️ Agenda inacessível no momento.</p>';
      return;
    }

    periods.forEach(p => {
      const card = document.createElement('div');
      card.className = `period-card ${p.disponivel ? 'available' : 'busy'}`;
      
      if (!p.disponivel) {
        card.innerHTML = `<strong>🚫 ${p.nome}</strong><br><small>${p.horario}</small><br><span style="font-size:12px;">👤 Ocupado por: ${p.ocupadoPor}</span>`;
      } else {
        card.innerHTML = `<strong>✅ ${p.nome}</strong><br><small>${p.horario}</small><br><span style="font-size:12px;">✨ Disponível para você</span>`;
        card.onclick = () => {
          card.classList.toggle('selected');
          if (selectedPeriods.includes(p.index)) {
            selectedPeriods = selectedPeriods.filter(i => i !== p.index);
          } else {
            selectedPeriods.push(p.index);
          }
          validate();
        };
      }
      resDiv.appendChild(card);
    });
  }

  function validate() {
    const p = document.getElementById('professor').value.trim();
    const t = document.getElementById('turma').value.trim();
    const h = document.getElementById('habilidades').value.trim();
    const d = document.getElementById('data').value;
    const btn = document.getElementById('submitBtn');
    btn.disabled = !(p && t && h && d && selectedPeriods.length > 0);
  }

  function save() {
    const btn = document.getElementById('submitBtn');
    btn.disabled = true;
    btn.innerText = "⏳ Gravando reserva...";

    const dados = {
      professor: document.getElementById('professor').value,
      turma: document.getElementById('turma').value,
      habilidades: document.getElementById('habilidades').value,
      data: document.getElementById('data').value,
      periodosSelecionados: selectedPeriods
    };

    google.script.run.withSuccessHandler((res) => {
      if(res.sucesso) {
        Swal.fire({
          title: 'Sucesso!',
          text: 'Sua reserva foi gravada na agenda.',
          icon: 'success',
          confirmButtonColor: '#0061ff'
        }).then(() => location.reload());
      } else {
        Swal.fire('Erro', res.mensagem, 'error');
        btn.disabled = false;
        btn.innerText = "🚀 Confirmar Agendamento";
      }
    }).criarReserva(dados);
  }
</script>
